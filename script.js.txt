const target = document.getElementById('target');
const scoreDisplay = document.getElementById('score');
let score = 0;

// Função para mover a imagem para uma posição aleatória
function moveTarget() {
    const x = Math.random() * (window.innerWidth - 100); // Limita ao tamanho da tela
    const y = Math.random() * (window.innerHeight - 100);
    target.style.left = `${x}px`;
    target.style.top = `${y}px`;
}

// Função para aumentar a pontuação e mover a imagem
target.addEventListener('click', () => {
    score++;
    scoreDisplay.textContent = score;
    moveTarget();
});

// Inicia o jogo movendo a imagem para a primeira posição
moveTarget();
